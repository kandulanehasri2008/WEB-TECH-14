/* =====================================================
   Meridian Diagnostics – Interactive components (script.js)
   Interactive component 1: search and filter for tests
   Interactive component 2: FAQ accordion

   Uses: addEventListener, querySelector, querySelectorAll,
         classList.toggle / add / remove, filter, forEach, find
   ===================================================== */
(function () {

  /* ---------- 1. SEARCH AND FILTER ---------- */
  const searchInput    = document.querySelector("#test-search");
  const categorySelect = document.querySelector("#test-category");
  const fastingSelect  = document.querySelector("#test-fasting");
  const clearButton    = document.querySelector("#test-clear");
  const countText      = document.querySelector("#test-count");
  const emptyText      = document.querySelector("#test-empty");

  /* querySelectorAll gives a NodeList; Array.from lets us use filter() and find() */
  const cards = Array.from(document.querySelectorAll("#tests .card-grid .card"));

  function cardName(card) {
    return card.querySelector("h3").textContent.toLowerCase();
  }

  function applyFilters() {
    const term     = searchInput.value.trim().toLowerCase();
    const category = categorySelect.value;
    const fasting  = fastingSelect.value;

    /* Keep only the cards that match all three controls */
    const visible = cards.filter(function (card) {
      const matchesText     = card.textContent.toLowerCase().includes(term);
      const matchesCategory = category === "all" || card.dataset.category === category;
      const matchesFasting  = fasting === "all" || card.dataset.fasting === fasting;
      return matchesText && matchesCategory && matchesFasting;
    });

    /* Show or hide each card */
    cards.forEach(function (card) {
      card.classList.toggle("is-hidden", !visible.includes(card));
      card.classList.remove("is-match");
    });

    /* find() returns the first card whose name is exactly what was typed */
    const exact = term ? visible.find(function (card) { return cardName(card) === term; }) : null;
    if (exact) {
      exact.classList.add("is-match");
    }

    countText.textContent = "Showing " + visible.length + " of " + cards.length + " tests";
    emptyText.hidden = visible.length > 0;
  }

  searchInput.addEventListener("input", applyFilters);
  categorySelect.addEventListener("change", applyFilters);
  fastingSelect.addEventListener("change", applyFilters);

  clearButton.addEventListener("click", function () {
    searchInput.value = "";
    categorySelect.value = "all";
    fastingSelect.value = "all";
    applyFilters();
    searchInput.focus();
  });

  applyFilters();

  /* ---------- 2. FAQ ACCORDION ---------- */
  const faqItems    = document.querySelectorAll(".faq-item");
  const expandAll   = document.querySelector("#faq-expand");
  const collapseAll = document.querySelector("#faq-collapse");
  let bulkChange = false;     /* true while Expand all / Collapse all runs */

  faqItems.forEach(function (item) {
    item.addEventListener("toggle", function () {
      item.classList.toggle("is-open", item.open);

      /* Opening one question closes the others */
      if (item.open && !bulkChange) {
        faqItems.forEach(function (other) {
          if (other !== item) {
            other.open = false;
          }
        });
      }
    });
  });

  expandAll.addEventListener("click", function () {
    bulkChange = true;
    faqItems.forEach(function (item) { item.open = true; });
    setTimeout(function () { bulkChange = false; }, 0);
  });

  collapseAll.addEventListener("click", function () {
    faqItems.forEach(function (item) { item.open = false; });
  });

})();
